"use strict"; document.addEventListener("DOMContentLoaded", function () { const startButton = document.getElementById("start"); const stopButton = document.getElementById("stop"); const testButton = document.getElementById("test"); const status = document.getElementById("status"); const call = document.getElementById("call"); console.log("ENGINEER.JS LOADED"); console.log("START:", startButton); console.log("STOP:", stopButton); console.log("TEST:", testButton); if (!startButton || !stopButton || !testButton) { console.error("ONE OR MORE BUTTONS WERE NOT FOUND"); return; } startButton.addEventListener("click", function () { console.log("START BUTTON WORKED"); status.textContent = "PUSH LAP"; call.textContent = "PUSH LAP"; if ("speechSynthesis" in window) { window.speechSynthesis.cancel(); const voice = new SpeechSynthesisUtterance( "Push lap. Build speed." ); voice.rate = 1.05; window.speechSynthesis.speak(voice); } }); stopButton.addEventListener("click", function () { console.log("STOP BUTTON WORKED"); status.textContent = "STOPPED"; call.textContent = "STOPPED"; if ("speechSynthesis" in window) { window.speechSynthesis.cancel(); } }); testButton.addEventListener("click", function () { console.log("TEST BUTTON WORKED"); status.textContent = "VOICE TEST"; call.textContent = "VOICE CHECK"; if ("speechSynthesis" in window) { window.speechSynthesis.cancel(); const voice = new SpeechSynthesisUtterance( "Engineer voice check." ); voice.rate = 1.05; window.speechSynthesis.speak(voice); } }); });
